/** * Bibliothèque CE - Backend de consultation du catalogue * -------------------------------------------------------- * Ce script doit être collé dans l'éditeur Apps Script LIÉ à votre Google Sheet * "AMB_CE_Bibliotheque" (menu Extensions > Apps Script depuis le Sheet). * * Il sert uniquement de point d'accès en lecture pour la page publique * catalogue.html (hébergée sur GitHub Pages) : celle-ci interroge cette URL * avec ?action=catalog pour récupérer la liste des livres en JSON. * * Le catalogue continue d'être rempli manuellement (copier-coller depuis les * exports de ton appli de scan personnelle), pas via un outil de scan intégré. */ function doGet(e) { if (e && e.parameter && e.parameter.action === 'catalog') { return ContentService.createTextOutput(JSON.stringify(getCatalog())) .setMimeType(ContentService.MimeType.JSON); } // Fallback : affiche une page simple si on ouvre l'URL directement dans un navigateur return HtmlService.createHtmlOutput( '

API Bibliothèque CE active. Utilise la page catalogue.html hébergée sur GitHub Pages pour consulter les livres.

' ); } /** * Renvoie le catalogue public (ISBN, Titre, Auteur, Type, Genre, Resume, Jaquette, Statut). * N'expose volontairement PAS les colonnes de réservation (nom/email/dates), * même si elles existent dans le Sheet. */ function getCatalog() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Catalogue'); if (!sheet) return []; var lastRow = sheet.getLastRow(); if (lastRow < 2) return []; var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0] .map(function(h) { return String(h).trim().toLowerCase(); }); var idx = { isbn: headers.indexOf('isbn'), titre: headers.indexOf('titre'), auteur: headers.indexOf('auteur'), type: headers.indexOf('type'), genre: headers.indexOf('genre'), tags: headers.indexOf('tags'), resume: headers.indexOf('resume'), resumeComplet: headers.indexOf('4e de couverture'), jaquette: headers.indexOf('jaquette (url)'), statut: headers.indexOf('statut'), datePublication: headers.indexOf('date de publication'), editeur: headers.indexOf('editeur'), pages: headers.indexOf('pages') }; var data = sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).getValues(); var out = []; for (var i = 0; i < data.length; i++) { var row = data[i]; if (!row[idx.isbn]) continue; // ligne vide out.push({ isbn: String(row[idx.isbn]), titre: row[idx.titre] || '', auteur: row[idx.auteur] || '', type: idx.type > -1 ? row[idx.type] || '' : '', genre: row[idx.genre] || '', tags: idx.tags > -1 ? row[idx.tags] || '' : '', resume: row[idx.resume] || '', resumeComplet: idx.resumeComplet > -1 ? row[idx.resumeComplet] || '' : '', jaquette: row[idx.jaquette] || '', statut: row[idx.statut] || '', datePublication: idx.datePublication > -1 ? row[idx.datePublication] || '' : '', editeur: idx.editeur > -1 ? row[idx.editeur] || '' : '', pages: idx.pages > -1 ? row[idx.pages] || '' : '' }); } return out; }
